# 11. FEATURE ENGINEERING
create_catboost_direct_features <- function(series, lags = 5) {
  n <- length(series)
  X <- matrix(NA, nrow = n - lags, ncol = lags + 1)
  y <- numeric(n - lags)
  
  for (i in (lags + 1):n) {
    idx <- i - lags
    X[idx, 1] <- i
    X[idx, 2:(lags + 1)] <- series[(i - 1):(i - lags)]
    y[idx] <- series[i]
  }
  
  colnames(X) <- c("time_idx", paste0("lag", 1:lags))
  list(X = X, y = y)
}


lags_cb <- 5


cb_direct_data <- create_catboost_direct_features(
  as.numeric(ts_affected),
  lags = lags_cb

# 14. FEATURE ENGINEERING ON RESIDUALS
create_residual_features <- function(residuals, lags = 5) {
  n <- length(residuals)
  X <- matrix(NA, nrow = n - lags, ncol = lags + 2)
  y <- numeric(n - lags)
  
  for (i in (lags + 1):n) {
    idx <- i - lags
    X[idx, 1] <- i
    X[idx, 2:(lags + 1)] <- residuals[(i - 1):(i - lags)]
    X[idx, lags + 2] <- mean(residuals[(i - lags):(i - 1)])
    y[idx] <- residuals[i]
  }
  
  colnames(X) <- c("time_idx", paste0("lag", 1:lags), "rolling_mean")
  list(X = X, y = y)
}

    mean(recent_residuals)

  pool <- catboost.load_pool(features)
  resid_pred <- catboost.predict(catboost_resid_model, pool)
  
  resid_pred <- resid_pred * DAMPING_FACTOR

  # update residual memory using actual
  actual_resid <- as.numeric(test_ts[i] - arima_pred_test[i])
  recent_residuals <- c(recent_residuals[-1], actual_resid)
}

calculate_metrics <- function(actual, predicted, model_name, naive = NULL) {
  
  mae <- mean(abs(actual - predicted))
  rmse <- sqrt(mean((actual - predicted)^2))
  mape <- mean(abs((actual - predicted) / actual)) * 100
  smape <- mean(2 * abs(actual - predicted) /
                  (abs(actual) + abs(predicted))) * 100
  
  ss_res <- sum((actual - predicted)^2)
  ss_tot <- sum((actual - mean(actual))^2)
  r2 <- 1 - ss_res / ss_tot
  
  if (!is.null(naive)) {
    theil_u <- sqrt(mean((actual - predicted)^2)) /
      sqrt(mean((actual - naive)^2))
  } else {
    theil_u <- NA
  }
  
  data.frame(
    Model = model_name,
    MAE = round(mae, 6),
    RMSE = round(rmse, 6),
    MAPE = round(mape, 4),
    SMAPE = round(smape, 4),
    R2 = round(r2, 6),
    Theil_U = round(theil_u, 6)
  )
}
